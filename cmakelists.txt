cmake_minimum_required(VERSION 4.2)

# 自动获取项目根目录文件夹名作为 PROJECT_NAME
get_filename_component(PROJECT_DIR_NAME "${CMAKE_SOURCE_DIR}" NAME)

project(${PROJECT_DIR_NAME} C CXX ASM)

add_subdirectory(StdPeriphDriver)
add_subdirectory(Bsp)

# ============================================================
# 通用固件目标规则：Boot / Dongle / Device 共用
# ============================================================
function(add_firmware TARGET_NAME MAIN_SOURCE ROLE_LIB LD_SCRIPT)
    if(NOT LD_SCRIPT)
        set(LD_SCRIPT ${CMAKE_CURRENT_SOURCE_DIR}/Link.ld)
    endif()

    add_executable(${TARGET_NAME}
        ${MAIN_SOURCE}
        ${CMAKE_CURRENT_SOURCE_DIR}/startup_CH572.S
    )

    # ---- 工程代码额外编译选项(对库代码不生效) ----
    target_compile_options(${TARGET_NAME} PRIVATE -Wall -Wextra)

    # ----指定 C / C++ 标准，与MRS2使用相同标准----
    set_target_properties(${TARGET_NAME}
        PROPERTIES
            C_STANDARD 99
            C_STANDARD_REQUIRED ON
            C_EXTENSIONS ON          # 对应 -std=gnu99
            CXX_STANDARD 11
            CXX_STANDARD_REQUIRED ON
            CXX_EXTENSIONS ON
    )

# ---- 链接脚本与库 ----
    target_link_options(${TARGET_NAME}
        PRIVATE
            -T${LD_SCRIPT}
            -Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/${TARGET_NAME}.map
    )

    target_link_libraries(${TARGET_NAME}
        PRIVATE
            ${ROLE_LIB}
            nluartlib
            ch57xstdlib
            ${CMAKE_CURRENT_SOURCE_DIR}/StdPeriphDriver/src/libISP572.a
            ${CMAKE_CURRENT_SOURCE_DIR}/StdPeriphDriver/src/libCH57xRF.a
    )
    set_target_properties(${TARGET_NAME} PROPERTIES SUFFIX ".elf")

    # ---- 构建后产物 ----
    set(FW_HEX ${CMAKE_CURRENT_BINARY_DIR}/${TARGET_NAME}.hex)
    set(FW_LST ${CMAKE_CURRENT_BINARY_DIR}/${TARGET_NAME}.lst)
    set(FW_BIN ${CMAKE_CURRENT_BINARY_DIR}/${TARGET_NAME}.bin)
    add_custom_command(TARGET ${TARGET_NAME}
        POST_BUILD
        COMMAND ${CMAKE_OBJCOPY} -O ihex   $<TARGET_FILE:${TARGET_NAME}> ${FW_HEX}
        COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${TARGET_NAME}> ${FW_BIN}
        COMMAND ${CMAKE_OBJDUMP} --source --all-headers --demangle -M xw
                                 --line-numbers --wide
                                 $<TARGET_FILE:${TARGET_NAME}> > ${FW_LST}
        COMMAND ${CMAKE_SIZE} --format=berkeley $<TARGET_FILE:${TARGET_NAME}>
        BYPRODUCTS ${FW_HEX} ${FW_LST} ${FW_BIN}
        COMMENT "Generating ${TARGET_NAME}.hex / .bin / .lst and size report"
        VERBATIM
    )
endfunction()

# ============================================================
# 固件 0：IAP Boot —— dongle(USB)/dev(RF+UART-IAP) 共用单一引导
# ============================================================
add_firmware(${PROJECT_NAME}_Boot
             ${CMAKE_CURRENT_SOURCE_DIR}/Core/Main_boot.c
             ""
             ${CMAKE_CURRENT_SOURCE_DIR}/Link_boot.ld)

# ---- boot.bin -> 可重定位目标文件，注入 App 镜像 0x0~0x8000 ----
set(BOOT_BIN ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}_Boot.bin)
set(IAP_BLOB_OBJ ${CMAKE_CURRENT_BINARY_DIR}/iap_blob.o)
add_custom_command(
    OUTPUT ${IAP_BLOB_OBJ}
    COMMAND ${CMAKE_OBJCOPY} -I binary -O elf32-littleriscv
            --rename-section .data=.iap,alloc,load,readonly,contents
            ${BOOT_BIN} ${IAP_BLOB_OBJ}
    DEPENDS ${PROJECT_NAME}_Boot
    COMMENT "Packing ${PROJECT_NAME}_Boot.bin into iap_blob.o (.iap)"
    VERBATIM
)
add_custom_target(iap_blob DEPENDS ${IAP_BLOB_OBJ})
set_source_files_properties(${IAP_BLOB_OBJ} PROPERTIES GENERATED TRUE)

# ============================================================
# 固件 1：Dongle —— PC 侧 USB 转 2.4G 无线串口适配器
# ============================================================
add_firmware(${PROJECT_NAME}_Dongle
             ${CMAKE_CURRENT_SOURCE_DIR}/Core/Main_dongle.c
             nluartdongle
             ${CMAKE_CURRENT_SOURCE_DIR}/Link.ld)
target_sources(${PROJECT_NAME}_Dongle PRIVATE ${IAP_BLOB_OBJ})
add_dependencies(${PROJECT_NAME}_Dongle iap_blob)

# Debug/RelWithDebInfo 下开调试打印
target_compile_definitions(${PROJECT_NAME}_Dongle PRIVATE
    $<$<OR:$<CONFIG:Debug>,$<CONFIG:RelWithDebInfo>>:DEBUG=1>
)

# ============================================================
# 固件 2：Device —— 接真实设备串口(PA0/PA1) + RESET/BOOT(PA2/PA3) 的一端
# ============================================================
add_firmware(${PROJECT_NAME}_Device
             ${CMAKE_CURRENT_SOURCE_DIR}/Core/Main_dev.c
             nluartdevice
             ${CMAKE_CURRENT_SOURCE_DIR}/Link.ld)
target_sources(${PROJECT_NAME}_Device PRIVATE ${IAP_BLOB_OBJ})
add_dependencies(${PROJECT_NAME}_Device iap_blob)

# ============================================================
# 后处理：App-only hex（不含 boot，供官方 WCHMcuIAP 工具升级用）
# ${TARGET}.hex 本身已含 boot+app，可直接烧录
# ============================================================
foreach(APP_TGT ${PROJECT_NAME}_Dongle ${PROJECT_NAME}_Device)
    add_custom_command(
        TARGET ${APP_TGT} POST_BUILD
        COMMAND ${CMAKE_OBJCOPY} -R .iap -O ihex
                $<TARGET_FILE:${APP_TGT}>
                ${CMAKE_CURRENT_BINARY_DIR}/${APP_TGT}_app.hex
        BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/${APP_TGT}_app.hex
        COMMENT "Generating ${APP_TGT}_app.hex (app-only, for WCHMcuIAP)"
        VERBATIM
    )
endforeach()
